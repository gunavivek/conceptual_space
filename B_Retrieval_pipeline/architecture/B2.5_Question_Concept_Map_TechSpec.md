### Key Algorithm Implementations

#### 1. Fuzzy Membership Calculation

```python
def calculate_fuzzy_membership(self, question_features: Dict, concept_definition: Dict) -> float:
    """
    Implement fuzzy membership function: μ_concept(q) = exp(-d²/2σ²)

    Architecture Reference: Core Design Decision in B2.5_QUESTION_CONCEPT_MAPPING.md

    Args:
        question_features: Aggregated question representation
        concept_definition: A2.4 concept data

    Returns:
        float: Fuzzy membership score (0.0-1.0)
    """

    # Extract concept parameters
    sigma = concept_definition["importance_score"] * self.config["importance_scaling"]
    concept_keywords = concept_definition["primary_keywords"]
    concept_weights = concept_definition["keyword_frequencies"]

    # Calculate distance components
    keyword_distance = self._compute_keyword_distance(
        question_features["primary_keywords"],
        concept_keywords,
        concept_weights
    )

    semantic_distance = self._compute_semantic_distance(
        question_features["answer_type"],
        concept_definition["canonical_name"]
    )

    temporal_distance = self._compute_temporal_distance(
        question_features.get("temporal_context", {}),
        concept_definition
    )

    # Weighted distance combination
    weights = self.config["distance_weights"]
    total_distance = (
        keyword_distance * weights["keyword_overlap"] +
        semantic_distance * weights["semantic_similarity"] +
        temporal_distance * weights.get("temporal_alignment", 0.1)
    )

    # Fuzzy membership calculation
    membership = math.exp(-(total_distance ** 2) / (2 * sigma ** 2))

    return min(1.0, max(0.0, membership))  # Clamp to [0,1]
```

#### 2. Feature Aggregation

```python
def aggregate_question_features(self, question_data: Dict) -> Dict[str, Any]:
    """
    Implement multi-source feature fusion strategy from architecture

    Args:
        question_data: Combined B2.x outputs

    Returns:
        dict: Unified question representation
    """

    # Primary signals (required)
    b21_data = question_data.get("b21", {})
    b23_data = question_data.get("b23", {})

    # Enhancement signals (optional)
    b22_data = question_data.get("b22", {})
    b24_data = question_data.get("b24", {})

    # Merge keywords with weights
    primary_keywords = b21_data.get("intent_keywords", [])
    keyword_weights = b21_data.get("keyword_weights", {})

    # Add declarative entities if available
    if b22_data:
        declarative_entities = b22_data.get("declarative_elements", {}).get("entities", [])
        primary_keywords.extend(declarative_entities)

    # Extract answer context
    answer_context = b23_data.get("answer_prediction", {})
    answer_type = answer_context.get("primary_type", "text")

    # Extract temporal context if available
    temporal_context = {}
    if b24_data:
        temporal_context = b24_data.get("temporal_context", {})

    # Calculate feature importance weights
    feature_weights = self._calculate_feature_importance(b21_data, b23_data, b24_data)

    return {
        "primary_keywords": list(set(primary_keywords)),  # Remove duplicates
        "keyword_weights": keyword_weights,
        "answer_type": answer_type,
        "answer_context": answer_context,
        "temporal_context": temporal_context,
        "feature_weights": feature_weights,
        "input_completeness": {
            "b21_available": bool(b21_data),
            "b22_available": bool(b22_data),
            "b23_available": bool(b23_data),
            "b24_available": bool(b24_data)
        }
    }
```

#### 3. Distance Calculation Functions

```python
def _compute_keyword_distance(self, question_keywords: List[str],
                            concept_keywords: List[str],
                            concept_weights: Dict[str, int]) -> float:
    """
    Calculate keyword-based distance between question and concept

    Returns:
        float: Distance score (0.0 = perfect match, 1.0 = no match)
    """

    if not question_keywords or not concept_keywords:
        return 1.0

    # Calculate weighted overlap
    question_set = set(kw.lower() for kw in question_keywords)
    concept_set = set(kw.lower() for kw in concept_keywords)

    overlap = question_set.intersection(concept_set)

    if not overlap:
        return 1.0

    # Weight overlap by concept keyword frequencies
    weighted_overlap = sum(concept_weights.get(kw, 1) for kw in overlap)
    total_concept_weight = sum(concept_weights.values())

    overlap_ratio = weighted_overlap / max(total_concept_weight, 1)

    return 1.0 - overlap_ratio

def _compute_semantic_distance(self, answer_type: str, concept_name: str) -> float:
    """
    Calculate semantic distance between expected answer type and concept

    Returns:
        float: Distance score (0.0 = highly relevant, 1.0 = irrelevant)
    """

    # Define answer type to concept relevance mappings
    type_concept_mappings = {
        "percentage": ["change", "ratio", "metrics", "performance"],
        "numeric": ["amount", "count", "value", "total"],
        "date": ["time", "period", "schedule", "timeline"],
        "text": ["description", "policy", "procedure", "definition"]
    }

    relevant_concepts = type_concept_mappings.get(answer_type.lower(), [])

    concept_lower = concept_name.lower()

    # Check for direct relevance
    for relevant in relevant_concepts:
        if relevant in concept_lower:
            return 0.2  # High relevance

    # Check for partial relevance (broader matching)
    if answer_type.lower() in ["percentage", "numeric"]:
        financial_terms = ["revenue", "cost", "income", "expense", "profit"]
        if any(term in concept_lower for term in financial_terms):
            return 0.4  # Medium relevance

    return 0.8  # Low relevance
```

## Error Handling and Graceful Degradation

### Input Validation Strategy

```python
def _validate_inputs(self, question_id: str) -> Dict[str, Any]:
    """
    Validate and load all required and optional inputs

    Returns:
        dict: Validated input data with status flags
    """

    validation_result = {
        "valid": True,
        "errors": [],
        "warnings": [],
        "data": {}
    }

    # Required inputs
    required_files = {
        "b21": self.config["file_paths"]["b21_output"],
        "b23": self.config["file_paths"]["b23_output"],
        "a24": self.config["file_paths"]["a24_concepts"]
    }

    for key, file_path in required_files.items():
        try:
            data = self._load_json_file(file_path)
            # Validate question_id exists in data
            if self._find_question_data(data, question_id):
                validation_result["data"][key] = data
            else:
                validation_result["errors"].append(f"Question {question_id} not found in {key}")
                validation_result["valid"] = False
        except FileNotFoundError:
            validation_result["errors"].append(f"Required file not found: {file_path}")
            validation_result["valid"] = False
        except json.JSONDecodeError:
            validation_result["errors"].append(f"Invalid JSON in file: {file_path}")
            validation_result["valid"] = False

    # Optional inputs (warnings only)
    optional_files = {
        "b22": self.config["file_paths"]["b22_output"],
        "b24": self.config["file_paths"]["b24_output"],
        "a25": self.config["file_paths"]["a25_concepts"]
    }

    for key, file_path in optional_files.items():
        try:
            data = self._load_json_file(file_path)
            validation_result["data"][key] = data
        except (FileNotFoundError, json.JSONDecodeError) as e:
            validation_result["warnings"].append(f"Optional file issue: {file_path} - {str(e)}")

    return validation_result

def _handle_missing_inputs(self, input_status: Dict) -> Dict[str, Any]:
    """
    Implement graceful degradation for missing optional inputs

    Args:
        input_status: Validation results from _validate_inputs

    Returns:
        dict: Question data with fallbacks for missing inputs
    """

    question_data = input_status["data"]

    # Handle missing B2.2 (declarative)
    if "b22" not in question_data:
        self.processing_stats["b22_fallback"] = "basic_keyword_extraction"

    # Handle missing B2.4 (temporal)
    if "b24" not in question_data:
        self.processing_stats["b24_fallback"] = "no_temporal_context"

    # Handle missing A2.5 (expanded concepts)
    if "a25" not in question_data:
        self.processing_stats["a25_fallback"] = "core_concepts_only"

    return question_data
```

## Performance Specifications

### Computational Complexity
- **Time Complexity**: O(n × m) where n = number of concepts, m = number of question features
- **Space Complexity**: O(n) for concept storage and membership calculations
- **Target Processing Time**: < 25ms per question

### Performance Monitoring

```python
class PerformanceMonitor:
    """Track and log performance metrics"""

    def __init__(self):
        self.start_time = None
        self.checkpoints = {}

    def start_timing(self):
        """Start performance measurement"""
        self.start_time = time.perf_counter()

    def checkpoint(self, name: str):
        """Record timing checkpoint"""
        if self.start_time:
            elapsed = (time.perf_counter() - self.start_time) * 1000  # ms
            self.checkpoints[name] = elapsed

    def get_metrics(self) -> Dict[str, float]:
        """Get all performance metrics"""
        total_time = max(self.checkpoints.values()) if self.checkpoints else 0
        return {
            "total_processing_time_ms": total_time,
            **{f"{name}_time_ms": time for name, time in self.checkpoints.items()}
        }
```
