# Q2.2: Enhanced Keyword Extraction - Architecture Specification

## Core Purpose

Q2.2 is the **semantic keyword extraction module** that identifies and weights critical terms from questions for geometric positioning. Unlike traditional keyword extraction, Q2.2 focuses on **domain-specific terminology**, **semantic importance**, and **geometric relevance** that contribute to coordinate calculation in the concept space.

## Key Innovation: Semantic Keyword Intelligence

Q2.2 introduces enhanced keyword extraction that goes beyond simple term frequency, providing **semantically weighted keywords** that directly influence geometric positioning in the Q-Pipeline.

## Core Architecture Components

### 1. Domain-Specific Term Extractor

```python
class DomainTermExtractor:
    """
    Extracts domain-specific terminology from questions
    """

    def __init__(self):
        self.financial_terms = {
            'revenue': 1.0, 'income': 0.9, 'expense': 0.9, 'profit': 0.9,
            'loss': 0.9, 'cost': 0.8, 'asset': 0.8, 'liability': 0.8,
            'equity': 0.8, 'cash': 0.7, 'investment': 0.7, 'dividend': 0.7
        }

        self.temporal_terms = {
            'year': 0.9, 'quarter': 0.8, 'month': 0.7, 'annual': 0.8,
            'fiscal': 0.8, 'period': 0.7, 'date': 0.6
        }

        self.quantitative_terms = {
            'percentage': 1.0, 'percent': 1.0, 'ratio': 0.9, 'rate': 0.8,
            'change': 0.8, 'growth': 0.8, 'increase': 0.7, 'decrease': 0.7,
            'total': 0.7, 'sum': 0.6, 'average': 0.6, 'difference': 0.7
        }

    def extract_domain_terms(self, question_text: str) -> List[Dict]:
        """
        Extract and weight domain-specific terms
        """
        text_lower = question_text.lower()
        domain_terms = []

        # Extract financial terms
        for term, weight in self.financial_terms.items():
            if term in text_lower:
                domain_terms.append({
                    'term': term,
                    'domain': 'financial',
                    'importance': weight,
                    'normalized_form': self._normalize_term(term)
                })

        # Extract temporal terms
        for term, weight in self.temporal_terms.items():
            if term in text_lower:
                domain_terms.append({
                    'term': term,
                    'domain': 'temporal',
                    'importance': weight,
                    'normalized_form': self._normalize_term(term)
                })

        # Extract quantitative terms
        for term, weight in self.quantitative_terms.items():
            if term in text_lower:
                domain_terms.append({
                    'term': term,
                    'domain': 'quantitative',
                    'importance': weight,
                    'normalized_form': self._normalize_term(term)
                })

        return domain_terms

    def _normalize_term(self, term: str) -> str:
        """
        Normalize term to canonical form
        """
        # Simple normalization - in practice would use lemmatization
        normalization_map = {
            'revenues': 'revenue',
            'incomes': 'income',
            'expenses': 'expense',
            'percentage': 'percent',
            'yearly': 'annual'
        }
        return normalization_map.get(term, term)
```

### 2. Semantic Keyword Analyzer

```python
class SemanticKeywordAnalyzer:
    """
    Analyzes semantic importance of keywords
    """

    def __init__(self):
        self.semantic_weights = {
            'interrogative': 0.3,  # Question words
            'noun': 0.8,          # Nouns are typically important
            'verb': 0.7,          # Action words
            'adjective': 0.5,     # Modifiers
            'number': 0.9         # Numbers are critical
        }

    def extract_semantic_keywords(self, question_text: str) -> List[Dict]:
        """
        Extract keywords with semantic importance scores
        """
        tokens = question_text.split()
        keywords = []

        for token in tokens:
            # Skip common stopwords
            if token.lower() in ['the', 'a', 'an', 'in', 'on', 'at', 'to', 'for']:
                continue

            # Determine semantic properties
            keyword_info = {
                'keyword': token.lower(),
                'weight': self._calculate_semantic_weight(token),
                'category': self._categorize_keyword(token),
                'pos_tag': self._get_pos_tag(token),
                'semantic_role': self._determine_semantic_role(token, question_text)
            }

            if keyword_info['weight'] > 0.3:  # Threshold for inclusion
                keywords.append(keyword_info)

        return keywords

    def _calculate_semantic_weight(self, token: str) -> float:
        """
        Calculate semantic importance weight
        """
        # Check if number
        if any(char.isdigit() for char in token):
            return 0.9

        # Check for interrogatives
        if token.lower() in ['what', 'when', 'where', 'who', 'why', 'how']:
            return 0.3

        # Default weight based on length and capitalization
        base_weight = min(1.0, len(token) / 10.0)
        if token[0].isupper():
            base_weight += 0.2

        return min(1.0, base_weight)

    def _categorize_keyword(self, token: str) -> str:
        """
        Categorize keyword type
        """
        if any(char.isdigit() for char in token):
            return 'entity'
        elif token.lower() in ['what', 'when', 'where', 'who', 'why', 'how']:
            return 'interrogative'
        elif token.lower() in ['is', 'was', 'are', 'were', 'change', 'increase', 'decrease']:
            return 'relational'
        else:
            return 'concept'
```

### 3. Entity Keyword Recognizer

```python
class EntityKeywordRecognizer:
    """
    Recognizes named entities and special keywords
    """

    def extract_entity_keywords(self, question_text: str) -> List[Dict]:
        """
        Extract entity keywords from question
        """
        entities = []

        # Extract year entities
        import re
        year_pattern = r'\b(19|20)\d{2}\b'
        for match in re.finditer(year_pattern, question_text):
            entities.append({
                'entity': match.group(),
                'entity_type': 'temporal',
                'confidence': 0.95,
                'span': [match.start(), match.end()]
            })

        # Extract percentage/number entities
        number_pattern = r'\b\d+\.?\d*%?\b'
        for match in re.finditer(number_pattern, question_text):
            entities.append({
                'entity': match.group(),
                'entity_type': 'numeric',
                'confidence': 0.9,
                'span': [match.start(), match.end()]
            })

        # Extract financial entities (simplified)
        financial_entities = ['revenue', 'income', 'profit', 'expense']
        text_lower = question_text.lower()
        for entity in financial_entities:
            if entity in text_lower:
                start = text_lower.index(entity)
                entities.append({
                    'entity': entity,
                    'entity_type': 'financial_concept',
                    'confidence': 0.85,
                    'span': [start, start + len(entity)]
                })

        return entities
```

### 4. Keyword Embedding Generator

```python
class KeywordEmbeddingGenerator:
    """
    Generates embedding vectors for keywords
    """

    def __init__(self):
        self.embedding_dim = 50
        # In practice, would use pre-trained embeddings (Word2Vec, GloVe, etc.)

    def generate_embeddings(self, keywords: List[Dict]) -> Dict:
        """
        Generate embedding vectors for keywords
        """
        keyword_vectors = {}

        for kw_info in keywords:
            keyword = kw_info['keyword']
            # Generate pseudo-embedding based on keyword properties
            embedding = self._create_keyword_embedding(keyword, kw_info)
            keyword_vectors[keyword] = embedding

        # Create aggregate embedding
        aggregate_vector = self._aggregate_embeddings(list(keyword_vectors.values()))

        return {
            'aggregate_vector': aggregate_vector,
            'keyword_vectors': keyword_vectors,
            'vector_dimension': self.embedding_dim
        }

    def _create_keyword_embedding(self, keyword: str, kw_info: Dict) -> List[float]:
        """
        Create embedding vector for keyword (simplified)
        """
        import hashlib
        import numpy as np

        # Use hash for reproducible pseudo-random embedding
        hash_val = int(hashlib.md5(keyword.encode()).hexdigest()[:8], 16)
        np.random.seed(hash_val)

        # Base embedding
        embedding = np.random.randn(self.embedding_dim) * 0.1

        # Adjust based on keyword properties
        if kw_info['category'] == 'domain':
            embedding[0:10] += 0.5  # Boost first dimensions for domain terms
        if kw_info['weight'] > 0.7:
            embedding = embedding * kw_info['weight']  # Scale by importance

        return embedding.tolist()

    def _aggregate_embeddings(self, embeddings: List[List[float]]) -> List[float]:
        """
        Aggregate multiple embeddings into single vector
        """
        import numpy as np

        if not embeddings:
            return [0.0] * self.embedding_dim

        # Simple mean aggregation
        aggregate = np.mean(embeddings, axis=0)
        return aggregate.tolist()
```

### 5. Semantic Clustering Engine

```python
class SemanticClusteringEngine:
    """
    Clusters keywords by semantic similarity
    """

    def cluster_keywords(self, keywords: List[Dict]) -> List[Dict]:
        """
        Group keywords into semantic clusters
        """
        # Predefined semantic clusters for financial QA
        cluster_definitions = {
            'temporal_cluster': ['year', 'period', 'quarter', 'annual', '2018', '2019'],
            'financial_cluster': ['revenue', 'income', 'expense', 'profit', 'cost'],
            'quantitative_cluster': ['percentage', 'change', 'growth', 'increase', 'total'],
            'query_cluster': ['what', 'how', 'when', 'where', 'why']
        }

        clusters = []
        for cluster_id, cluster_keywords in cluster_definitions.items():
            matched_keywords = []
            for kw_info in keywords:
                if any(ck in kw_info['keyword'].lower() for ck in cluster_keywords):
                    matched_keywords.append(kw_info['keyword'])

            if matched_keywords:
                clusters.append({
                    'cluster_id': cluster_id,
                    'cluster_theme': cluster_id.replace('_cluster', ''),
                    'keywords': matched_keywords,
                    'coherence_score': len(matched_keywords) / len(cluster_keywords)
                })

        return clusters
```

### Step 2: Comprehensive Keyword Extraction
```python
def extract_enhanced_keywords(question_text: str) -> Dict:
    """
    Perform comprehensive keyword extraction
    """
    # Extract domain-specific terms
    domain_terms = DomainTermExtractor().extract_domain_terms(question_text)

    # Extract semantic keywords
    semantic_keywords = SemanticKeywordAnalyzer().extract_semantic_keywords(question_text)

    # Extract entity keywords
    entity_keywords = EntityKeywordRecognizer().extract_entity_keywords(question_text)

    # Cluster keywords semantically
    all_keywords = semantic_keywords + [{'keyword': dt['term']} for dt in domain_terms]
    semantic_clusters = SemanticClusteringEngine().cluster_keywords(all_keywords)

    # Generate embeddings
    embeddings = KeywordEmbeddingGenerator().generate_embeddings(semantic_keywords)

    return {
        'primary_keywords': semantic_keywords,
        'domain_specific_terms': domain_terms,
        'entity_keywords': entity_keywords,
        'semantic_clusters': semantic_clusters,
        'keyword_embeddings': embeddings
    }
```

### Step 3: Feature Calculation
```python
def calculate_keyword_features(extraction_results: Dict) -> Dict:
    """
    Calculate keyword feature statistics
    """
    total_keywords = len(extraction_results['primary_keywords'])
    unique_keywords = len(set(k['keyword'] for k in extraction_results['primary_keywords']))
    domain_terms = len(extraction_results['domain_specific_terms'])

    return {
        'total_keywords': total_keywords,
        'unique_keywords': unique_keywords,
        'domain_term_ratio': domain_terms / max(1, total_keywords),
        'semantic_density': unique_keywords / max(1, total_keywords),
        'keyword_diversity': 1.0 - (1.0 / max(1, unique_keywords))
    }
```
