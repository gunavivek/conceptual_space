# Q2.5: Document-Specific Convex Ball Assignment - Technical Specification

## Critical Innovation

Q2.5 is the **revolutionary cornerstone** of the Q-Pipeline, establishing the mathematical bridge between questions and document-specific concept spaces. Unlike B2.5 which uses global similarity, Q2.5 assigns questions to the **exact same coordinate system** used by A-Pipeline chunks.

def assign_to_ball(point: np.array, ball: ConvexBall) -> dict:
    distance = np.linalg.norm(point - ball.centroid)
    if distance <= ball.radius:
        return {
            "is_member": True,
            "distance_to_centroid": distance,
            "membership_strength": 1 - (distance / ball.radius),
            "position_vector": point - ball.centroid
        }
    return {"is_member": False}

## Core Algorithm

### Step 1: Load Document-Specific Concept Space
```python
def load_concept_space(doc_id: str) -> dict:
    """
    Load the EXACT concept space used by A-Pipeline for this document
    """
    # Load A-Pipeline outputs
    chunks_data = json.load(open(f"A_Concept_pipeline/outputs/A3_multi_strategy_chunks.json"))
    concepts_data = json.load(open(f"A_Concept_pipeline/outputs/A2.4_core_concepts.json"))

    # Filter to specific document
    doc_chunks = [c for c in chunks_data if c["doc_id"] == doc_id]
    doc_concepts = concepts_data.get(doc_id, {})

    # Extract concept centroids
    centroids = {}
    for concept_id, concept_data in doc_concepts.items():
        centroids[concept_id] = {
            "vector": np.array(concept_data["centroid"]),
            "radius": concept_data.get("radius", 1.0),
            "importance": concept_data.get("importance", 1.0)
        }

    return {
        "centroids": centroids,
        "dimensions": len(next(iter(centroids.values()))["vector"]),
        "chunk_count": len(doc_chunks),
        "concept_labels": list(centroids.keys())
    }
```

### Step 2: Calculate Question Coordinates
```python
def calculate_question_coordinates(question_data: dict, concept_space: dict) -> np.array:
    """
    Map question to n-dimensional coordinates using concept centroids
    """
    dimensions = concept_space["dimensions"]
    coordinates = np.zeros(dimensions)

    # Weight contributions from different layers
    weights = {
        "intent": 0.3,
        "semantic": 0.4,
        "context": 0.2,
        "temporal": 0.1
    }

    # Intent contribution
    if question_data.get("q21_intent"):
        intent_vector = vectorize_intent(question_data["q21_intent"], dimensions)
        coordinates += weights["intent"] * intent_vector

    # Semantic contribution (PRIMARY)
    if question_data.get("q22_semantic"):
        semantic_vector = map_to_concept_space(
            question_data["q22_semantic"],
            concept_space["centroids"]
        )
        coordinates += weights["semantic"] * semantic_vector

    # Context contribution
    if question_data.get("q23_context"):
        context_vector = encode_document_alignment(
            question_data["q23_context"],
            dimensions
        )
        coordinates += weights["context"] * context_vector

    # Temporal contribution
    if question_data.get("q24_temporal"):
        temporal_vector = encode_temporal_position(
            question_data["q24_temporal"],
            dimensions
        )
        coordinates += weights["temporal"] * temporal_vector

    # Normalize to unit hypersphere
    return coordinates / np.linalg.norm(coordinates)
```

### Step 3: Assign to Convex Balls
```python
def assign_to_convex_balls(coordinates: np.array, concept_space: dict) -> List[dict]:
    """
    Determine which convex balls the question belongs to
    """
    assignments = []

    for concept_id, centroid_data in concept_space["centroids"].items():
        centroid = centroid_data["vector"]
        radius = centroid_data["radius"]

        # Calculate Euclidean distance
        distance = np.linalg.norm(coordinates - centroid)

        if distance <= radius:
            # Question is within this convex ball
            assignments.append({
                "ball_id": concept_id,
                "centroid": centroid.tolist(),
                "radius": radius,
                "distance_to_centroid": distance,
                "membership_strength": 1 - (distance / radius),
                "importance_weight": centroid_data["importance"]
            })

    # Sort by membership strength
    assignments.sort(key=lambda x: x["membership_strength"], reverse=True)
    return assignments
```

### Step 4: Validate Geometric Consistency
```python
def validate_assignments(assignments: List[dict], coordinates: np.array) -> dict:
    """
    Ensure geometric consistency of assignments
    """
    validation = {
        "is_valid": True,
        "warnings": [],
        "metrics": {}
    }

    # Check if question assigned to at least one ball
    if len(assignments) == 0:
        validation["is_valid"] = False
        validation["warnings"].append("Question not assigned to any convex ball")

    # Check triangle inequality between assigned balls
    if len(assignments) >= 2:
        for i, ball1 in enumerate(assignments):
            for ball2 in assignments[i+1:]:
                centroid_dist = np.linalg.norm(
                    np.array(ball1["centroid"]) - np.array(ball2["centroid"])
                )
                if centroid_dist > ball1["radius"] + ball2["radius"]:
                    # Balls don't overlap but question assigned to both
                    validation["warnings"].append(
                        f"Geometric inconsistency: non-overlapping balls {ball1['ball_id']}, {ball2['ball_id']}"
                    )

    # Calculate coverage metrics
    validation["metrics"]["total_assignments"] = len(assignments)
    validation["metrics"]["max_membership"] = max(a["membership_strength"] for a in assignments) if assignments else 0
    validation["metrics"]["avg_distance"] = np.mean([a["distance_to_centroid"] for a in assignments]) if assignments else float('inf')

    return validation
```

## Performance Considerations

### Optimization Strategies
1. **Cache concept spaces** per document (don't reload for each question)
2. **Vectorize operations** using NumPy for coordinate calculations
3. **Pre-compute** centroid distances for frequently accessed documents
4. **Batch process** questions from same document together

### Expected Performance
- **Processing time**: <100ms per question
- **Memory usage**: ~50MB for concept space cache
- **Assignment accuracy**: >95% questions assigned to at least one ball
- **Average assignments**: 2-4 balls per question

## Error Handling

```python
def handle_assignment_failure(question_data: dict, error: Exception) -> dict:
    """
    Graceful degradation when assignment fails
    """
    fallback = {
        "question_id": question_data["question_id"],
        "doc_id": question_data["doc_id"],
        "error": str(error),
        "fallback_strategy": "nearest_centroid",
        "coordinates": None,
        "convex_ball_assignments": []
    }

    try:
        # Attempt simpler nearest-centroid assignment
        concept_space = load_concept_space(question_data["doc_id"])
        simple_coords = simple_semantic_mapping(question_data["question_text"])
        nearest = find_nearest_centroid(simple_coords, concept_space)
        fallback["convex_ball_assignments"] = [nearest]
    except:
        # Complete failure - mark for manual review
        fallback["requires_manual_review"] = True

    return fallback
```
