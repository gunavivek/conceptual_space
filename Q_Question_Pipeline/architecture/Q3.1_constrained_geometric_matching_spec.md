## Mathematical Framework

### Constraint Function
```python
def satisfies_convex_ball_constraint(question_balls: Set[str], chunk_balls: Set[str]) -> bool:
    """
    Core constraint: Match is valid IFF question and chunk share at least one ball
    """
    return len(question_balls.intersection(chunk_balls)) > 0
```

### Distance Calculation Within Constraints
```python
def constrained_euclidean_distance(q_coords: np.array, c_coords: np.array, shared_balls: List[dict]) -> float:
    """
    Calculate distance with convex ball weighting
    """
    base_distance = np.linalg.norm(q_coords - c_coords)

    # Weight by shared ball characteristics
    weight = 1.0
    for ball in shared_balls:
        # Closer to shared centroid = stronger match
        q_dist_to_centroid = np.linalg.norm(q_coords - ball["centroid"])
        c_dist_to_centroid = np.linalg.norm(c_coords - ball["centroid"])
        centroid_alignment = 1 - abs(q_dist_to_centroid - c_dist_to_centroid) / ball["radius"]
        weight *= (1 + centroid_alignment * 0.5)

    return base_distance / weight
```

### Step 2: Apply Convex Ball Constraints
```python
def filter_by_convex_ball_constraint(question_data: dict, chunks: List[dict]) -> List[dict]:
    """
    CRITICAL: Filter chunks to only those sharing convex balls with question
    """
    # Extract question's ball memberships
    question_balls = {
        ball["ball_id"]
        for ball in question_data["convex_ball_assignments"]
        if ball["membership_strength"] >= 0.3  # Minimum threshold
    }

    eligible_chunks = []
    for chunk in chunks:
        # Extract chunk's ball memberships
        chunk_balls = set(chunk.get("convex_balls", []))

        # Check constraint
        shared_balls = question_balls.intersection(chunk_balls)
        if shared_balls:
            # Chunk passes constraint
            chunk["shared_balls_with_question"] = list(shared_balls)
            chunk["constraint_strength"] = len(shared_balls) / len(question_balls)
            eligible_chunks.append(chunk)

    return eligible_chunks
```

### Step 3: Calculate Constrained Geometric Distances
```python
def calculate_geometric_distances(question_data: dict, eligible_chunks: List[dict]) -> List[dict]:
    """
    Calculate distances only for constraint-passing chunks
    """
    q_coords = np.array(question_data["coordinates"])
    q_intent = np.array(question_data.get("intent_vector", np.zeros(len(q_coords))))

    matches = []
    for chunk in eligible_chunks:
        c_coords = np.array(chunk["coordinates"])

        # Base Euclidean distance
        geometric_distance = np.linalg.norm(q_coords - c_coords)

        # Apply convex ball weighting
        weighted_distance = apply_convex_ball_weighting(
            geometric_distance,
            q_coords,
            c_coords,
            chunk["shared_balls_with_question"],
            question_data["convex_ball_assignments"]
        )

        # Calculate intent alignment
        if "capability_vector" in chunk:
            c_capability = np.array(chunk["capability_vector"])
            intent_alignment = np.dot(q_intent, c_capability) / (
                np.linalg.norm(q_intent) * np.linalg.norm(c_capability) + 1e-10
            )
        else:
            intent_alignment = 0.0

        matches.append({
            "chunk_id": chunk["chunk_id"],
            "chunk_text": chunk.get("text", ""),
            "geometric_distance": geometric_distance,
            "weighted_distance": weighted_distance,
            "shared_balls": chunk["shared_balls_with_question"],
            "constraint_strength": chunk["constraint_strength"],
            "intent_alignment": intent_alignment,
            "coordinates": c_coords.tolist()
        })

    return matches
```

### Step 4: Rank and Filter Results
```python
def rank_constrained_matches(matches: List[dict], max_chunks: int = 10) -> List[dict]:
    """
    Rank matches by weighted distance and constraint strength
    """
    # Calculate composite score
    for match in matches:
        # Prioritize: constraint strength > geometric proximity > intent alignment
        match["composite_score"] = (
            0.4 * match["constraint_strength"] +
            0.4 * (1 / (1 + match["weighted_distance"])) +
            0.2 * max(0, match["intent_alignment"])  # Only positive alignment
        )

    # Sort by composite score
    matches.sort(key=lambda x: x["composite_score"], reverse=True)

    # Add ranks
    for i, match in enumerate(matches):
        match["rank"] = i + 1

    return matches[:max_chunks]
```

### Step 5: Calculate Metrics
```python
def calculate_geometric_metrics(matches: List[dict], total_chunks: int) -> dict:
    """
    Calculate performance and quality metrics
    """
    if not matches:
        return {
            "search_space_reduction": {
                "total_chunks": total_chunks,
                "eligible_chunks": 0,
                "reduction_percentage": 100.0
            },
            "geometric_metrics": {}
        }

    distances = [m["geometric_distance"] for m in matches]
    constraints = [m["constraint_strength"] for m in matches]
    alignments = [m["intent_alignment"] for m in matches]

    return {
        "search_space_reduction": {
            "total_chunks": total_chunks,
            "eligible_chunks": len(matches),
            "reduction_percentage": (1 - len(matches) / total_chunks) * 100
        },
        "geometric_metrics": {
            "avg_distance": np.mean(distances),
            "distance_variance": np.var(distances),
            "min_distance": np.min(distances),
            "max_distance": np.max(distances),
            "avg_constraint_strength": np.mean(constraints),
            "avg_intent_alignment": np.mean(alignments),
            "positive_alignment_ratio": sum(1 for a in alignments if a > 0) / len(alignments)
        }
    }
```

## Optimization Techniques

### 1. Spatial Indexing
```python
class ConvexBallIndex:
    """
    Spatial index for fast constraint checking
    """
    def __init__(self, chunks: List[dict]):
        self.ball_to_chunks = defaultdict(list)
        for chunk in chunks:
            for ball_id in chunk.get("convex_balls", []):
                self.ball_to_chunks[ball_id].append(chunk["chunk_id"])

    def get_eligible_chunks(self, question_balls: Set[str]) -> Set[str]:
        eligible = set()
        for ball_id in question_balls:
            eligible.update(self.ball_to_chunks[ball_id])
        return eligible
```

### 2. Distance Caching
```python
class DistanceCache:
    """
    Cache computed distances for repeated queries
    """
    def __init__(self, cache_size: int = 1000):
        self.cache = {}
        self.cache_size = cache_size

    def get_distance(self, q_id: str, c_id: str) -> Optional[float]:
        key = f"{q_id}_{c_id}"
        return self.cache.get(key)

    def set_distance(self, q_id: str, c_id: str, distance: float):
        if len(self.cache) >= self.cache_size:
            # LRU eviction
            oldest = min(self.cache.keys())
            del self.cache[oldest]
        self.cache[f"{q_id}_{c_id}"] = distance
```

### 3. Batch Processing
```python
def batch_constrained_matching(questions: List[dict], doc_id: str) -> List[dict]:
    """
    Process multiple questions from same document efficiently
    """
    # Load document data once
    chunks = load_document_chunks(doc_id)
    spatial_index = ConvexBallIndex(chunks)

    results = []
    for question in questions:
        # Use spatial index for fast filtering
        question_balls = {b["ball_id"] for b in question["convex_ball_assignments"]}
        eligible_chunk_ids = spatial_index.get_eligible_chunks(question_balls)

        # Process only eligible chunks
        eligible_chunks = [c for c in chunks if c["chunk_id"] in eligible_chunk_ids]
        matches = calculate_geometric_distances(question, eligible_chunks)
        results.append(rank_constrained_matches(matches))

    return results
```
