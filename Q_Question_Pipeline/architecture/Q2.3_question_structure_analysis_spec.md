# Q2.3: Question Structure Analysis - Architecture Specification

## Core Purpose

Q2.3 is the **question structure intelligence module** that analyzes the linguistic and syntactic structure of questions. Unlike document analysis (A-Pipeline's responsibility), Q2.3 focuses on **question complexity**, **syntactic patterns**, and **linguistic dependencies** that inform geometric coordinate calculation.

## Key Innovation: Question Linguistic Intelligence

Q2.3 introduces revolutionary question structure analysis that enables the Q-Pipeline to understand **how questions are constructed linguistically**, providing critical structural features for geometric positioning.

## Core Architecture Components

### 1. Syntactic Feature Extractor

```python
class SyntacticFeatureExtractor:
    """
    Extracts syntactic features from question text
    """

    def __init__(self):
        self.nlp = spacy.load("en_core_web_sm")
        self.question_patterns = {
            'wh_questions': ['what', 'when', 'where', 'who', 'why', 'how'],
            'yes_no_questions': ['is', 'are', 'do', 'does', 'did', 'can', 'could', 'will', 'would'],
            'modal_verbs': ['can', 'could', 'should', 'would', 'might', 'may', 'must']
        }

    def extract_syntactic_features(self, question_text: str) -> Dict:
        """
        Extract comprehensive syntactic features
        """
        doc = self.nlp(question_text)

        # Question type classification
        question_type = self.classify_question_type(doc)

        # WH-word identification
        wh_word = self.identify_wh_word(doc)

        # Verb analysis
        auxiliary_verb, main_verb = self.analyze_verbs(doc)

        # Voice and tense analysis
        voice = self.analyze_voice(doc)
        tense = self.analyze_tense(doc)

        # Modality detection
        modality = self.detect_modality(doc)

        return {
            'question_type': question_type,
            'wh_word': wh_word,
            'auxiliary_verb': auxiliary_verb,
            'main_verb': main_verb,
            'voice': voice,
            'tense': tense,
            'modality': modality
        }
```

### 2. Complexity Analyzer

```python
class ComplexityAnalyzer:
    """
    Analyzes syntactic complexity of questions
    """

    def analyze_complexity(self, question_text: str) -> Dict:
        """
        Calculate comprehensive complexity metrics
        """
        doc = self.nlp(question_text)

        # Basic metrics
        token_count = len([token for token in doc if not token.is_punct])

        # Dependency depth calculation
        dependency_depth = self.calculate_dependency_depth(doc)

        # Clause counting
        clause_count = self.count_clauses(doc)

        # Modifier counting
        modifier_count = self.count_modifiers(doc)

        # Prepositional phrase counting
        prepositional_phrases = self.count_prepositional_phrases(doc)

        # Composite complexity score
        syntactic_complexity = self.calculate_complexity_score(
            token_count, dependency_depth, clause_count, modifier_count
        )

        return {
            'token_count': token_count,
            'dependency_depth': dependency_depth,
            'clause_count': clause_count,
            'modifier_count': modifier_count,
            'prepositional_phrases': prepositional_phrases,
            'syntactic_complexity_score': syntactic_complexity
        }
```

### 3. Dependency Structure Analyzer

```python
class DependencyStructureAnalyzer:
    """
    Analyzes dependency parse structure of questions
    """

    def analyze_dependency_structure(self, question_text: str) -> Dict:
        """
        Extract dependency structure information
        """
        doc = self.nlp(question_text)

        # Find root token
        root_token = self.find_root_token(doc)

        # Extract subject and object
        subject, obj = self.extract_subject_object(doc)

        # Build dependency relations
        dependency_relations = self.build_dependency_relations(doc)

        # Calculate parse tree depth
        parse_tree_depth = self.calculate_parse_tree_depth(doc)

        return {
            'root_token': root_token,
            'subject': subject,
            'object': obj,
            'dependency_relations': dependency_relations,
            'parse_tree_depth': parse_tree_depth
        }
```

### 4. Linguistic Pattern Extractor

```python
class LinguisticPatternExtractor:
    """
    Extracts linguistic patterns and structures
    """

    def extract_linguistic_patterns(self, question_text: str) -> Dict:
        """
        Extract comprehensive linguistic patterns
        """
        doc = self.nlp(question_text)

        # POS sequence
        pos_sequence = [token.pos_ for token in doc if not token.is_punct]

        # Phrase extraction
        noun_phrases = self.extract_noun_phrases(doc)
        verb_phrases = self.extract_verb_phrases(doc)

        # Named entity recognition
        named_entities = self.extract_named_entities(doc)

        # Grammatical function analysis
        grammatical_functions = self.analyze_grammatical_functions(doc)

        return {
            'pos_sequence': pos_sequence,
            'noun_phrases': noun_phrases,
            'verb_phrases': verb_phrases,
            'named_entities': named_entities,
            'grammatical_functions': grammatical_functions
        }
```

### 5. Ambiguity Assessor

```python
class AmbiguityAssessor:
    """
    Assesses various types of ambiguity in questions
    """

    def assess_ambiguity(self, question_text: str, linguistic_patterns: Dict) -> Dict:
        """
        Comprehensive ambiguity assessment
        """
        # Structural ambiguity (multiple parse interpretations)
        structural_ambiguity = self.assess_structural_ambiguity(question_text)

        # Referential ambiguity (unclear references)
        referential_ambiguity = self.assess_referential_ambiguity(linguistic_patterns)

        # Scope ambiguity (quantifier/negation scope)
        scope_ambiguity = self.assess_scope_ambiguity(question_text)

        # Overall clarity score
        overall_clarity = 1.0 - max(structural_ambiguity, referential_ambiguity, scope_ambiguity)

        return {
            'structural_ambiguity': structural_ambiguity,
            'referential_ambiguity': referential_ambiguity,
            'scope_ambiguity': scope_ambiguity,
            'overall_clarity_score': max(0.0, overall_clarity)
        }
```

## Implementation Algorithm

### Step 1: Question Linguistic Analysis
```python
def analyze_question_structure(question_id: str) -> Dict:
    """
    Comprehensive question structure analysis
    """
    # Load question from Q1 output
    question_data = load_question_from_q1(question_id)

    # Syntactic feature extraction
    syntactic_features = SyntacticFeatureExtractor().extract_syntactic_features(
        question_data['question_text']
    )

    # Complexity analysis
    complexity_metrics = ComplexityAnalyzer().analyze_complexity(
        question_data['question_text']
    )

    # Dependency structure analysis
    dependency_structure = DependencyStructureAnalyzer().analyze_dependency_structure(
        question_data['question_text']
    )

    # Linguistic pattern extraction
    linguistic_patterns = LinguisticPatternExtractor().extract_linguistic_patterns(
        question_data['question_text']
    )

    return {
        'syntactic_features': syntactic_features,
        'complexity_metrics': complexity_metrics,
        'dependency_structure': dependency_structure,
        'linguistic_patterns': linguistic_patterns
    }
```

### Step 2: Structural Feature Vector Generation
```python
def generate_structural_features_vector(structural_analysis: Dict) -> List[float]:
    """
    Generate geometric feature vector for concept space positioning
    """
    features = []

    # Question type encoding (1.0 for WH-questions, 0.0 for yes/no)
    features.append(1.0 if structural_analysis['syntactic_features']['question_type'] == 'wh_question' else 0.0)

    # WH-word specificity
    wh_specificity = calculate_wh_specificity(structural_analysis['syntactic_features']['wh_word'])
    features.append(wh_specificity)

    # Voice encoding (1.0 for active, 0.0 for passive)
    features.append(1.0 if structural_analysis['syntactic_features']['voice'] == 'active' else 0.0)

    # Normalized complexity score
    features.append(structural_analysis['complexity_metrics']['syntactic_complexity_score'])

    # Normalized dependency depth
    max_depth = 5.0  # Reasonable maximum for normalization
    normalized_depth = min(structural_analysis['dependency_structure']['parse_tree_depth'] / max_depth, 1.0)
    features.append(normalized_depth)

    # Additional features...
    features.extend([0.4, 1.0, 1.0, 1.0, 1.0])  # Placeholder for remaining dimensions

    return features
```
